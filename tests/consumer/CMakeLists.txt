cmake_minimum_required(VERSION 3.31)
project(iiVoiceOverConsumer LANGUAGES CXX)
enable_testing()
set(CMAKE_DISABLE_FIND_PACKAGE_iiSDKCore TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_iiFileProviderNative TRUE)
find_package(iiVoiceOver 0.2 CONFIG REQUIRED)
if(TARGET iiSDKCore::Contracts OR TARGET iiFileProvider::Native)
    message(FATAL_ERROR "Voice SDK must not import peer SDK targets")
endif()
add_executable(iiVoiceOver_consumer main.cpp)
target_link_libraries(iiVoiceOver_consumer PRIVATE iiVoiceOver::iiVoiceOver)
set_target_properties(iiVoiceOver_consumer PROPERTIES CXX_EXTENSIONS OFF)
get_filename_component(voice_prefix "${iiVoiceOver_DIR}/../../.." ABSOLUTE)
add_test(NAME iiVoiceOver.installed_consumer COMMAND iiVoiceOver_consumer "${voice_prefix}" "${CMAKE_CURRENT_BINARY_DIR}")
set_tests_properties(iiVoiceOver.installed_consumer PROPERTIES TIMEOUT 120)
